Downloads matplotlib, about 10 MB more.
The code that drew this figure

The last line returns the figure as text; that is how it leaves Python and shows up on the page.

import matplotlib
matplotlib.use("Agg")          # draw in memory, not in a window
import io, base64
import numpy as np
import matplotlib.pyplot as plt

rng = np.random.default_rng(7)
x = rng.uniform(0, 10, n)      # n comes from the slider above
y = 0.8 * x + rng.normal(0, 1.2, n)

fig, ax = plt.subplots(figsize=(6.4, 4))
ax.scatter(x, y, c="#7B61FF", s=36, edgecolor="white")
ax.set_xlabel("measurement")
ax.set_ylabel("answer")

buf = io.BytesIO()
fig.savefig(buf, format="png", dpi=120, bbox_inches="tight")
plt.close(fig)
base64.b64encode(buf.getvalue()).decode()
